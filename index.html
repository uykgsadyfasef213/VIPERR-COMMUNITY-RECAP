import asyncio import logging import os import sqlite3 from datetime import datetime from pathlib import Path from aiogram import Bot, Dispatcher, F from aiogram.client.default import DefaultBotProperties from aiogram.enums import ParseMode from aiogram.filters import CommandStart from aiogram.types import ( CallbackQuery, InlineKeyboardButton, InlineKeyboardMarkup, Message, ) from dotenv import load_dotenv # ========================================================= # НАСТРОЙКИ # ========================================================= load_dotenv() BOT_TOKEN = os.getenv("BOT_TOKEN") GROUP_ID = os.getenv("GROUP_ID") if not BOT_TOKEN: raise ValueError( "❌ Не найден BOT_TOKEN. " "Добавь его в переменные окружения." ) if not GROUP_ID: raise ValueError( "❌ Не найден GROUP_ID. " "Добавь ID Telegram-группы." ) GROUP_ID = int(GROUP_ID) DB_PATH = Path("viperr.db") # ========================================================= # ЛОГИ # ========================================================= logging.basicConfig( level=logging.INFO, format="%(asctime)s | %(levelname)s | %(message)s", ) logger = logging.getLogger(__name__) # ========================================================= # TELEGRAM BOT # ========================================================= bot = Bot( token=BOT_TOKEN, default=DefaultBotProperties( parse_mode=ParseMode.HTML ), ) dp = Dispatcher() # ========================================================= # DATABASE # ========================================================= def get_connection(): connection = sqlite3.connect(DB_PATH) connection.row_factory = sqlite3.Row return connection def init_database(): connection = get_connection() connection.execute( """ CREATE TABLE IF NOT EXISTS users ( user_id INTEGER PRIMARY KEY, username TEXT, full_name TEXT, messages INTEGER DEFAULT 0, first_seen TEXT, last_seen TEXT ) """ ) connection.execute( """ CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER NOT NULL, created_at TEXT NOT NULL ) """ ) connection.commit() connection.close() logger.info("Database initialized") def save_message( user_id: int, username: str | None, full_name: str, ): now = datetime.utcnow().isoformat() connection = get_connection() existing_user = connection.execute( """ SELECT user_id FROM users WHERE user_id = ? """, (user_id,), ).fetchone() if existing_user: connection.execute( """ UPDATE users SET username = ?, full_name = ?, messages = messages + 1, last_seen = ? WHERE user_id = ? """, ( username, full_name, now, user_id, ), ) else: connection.execute( """ INSERT INTO users ( user_id, username, full_name, messages, first_seen, last_seen ) VALUES (?, ?, ?, 1, ?, ?) """, ( user_id, username, full_name, now, now, ), ) connection.execute( """ INSERT INTO messages ( user_id, created_at ) VALUES (?, ?) """, ( user_id, now, ), ) connection.commit() connection.close() def get_statistics(): connection = get_connection() total_messages = connection.execute( """ SELECT COUNT(*) AS count FROM messages """ ).fetchone()["count"] total_users = connection.execute( """ SELECT COUNT(*) AS count FROM users """ ).fetchone()["count"] active_users = connection.execute( """ SELECT COUNT(*) AS count FROM users WHERE messages > 0 """ ).fetchone()["count"] top_users = connection.execute( """ SELECT full_name, username, messages FROM users ORDER BY messages DESC LIMIT 10 """ ).fetchall() best_month = connection.execute( """ SELECT strftime('%m', created_at) AS month, COUNT(*) AS messages FROM messages GROUP BY month ORDER BY messages DESC LIMIT 1 """ ).fetchone() best_day = connection.execute( """ SELECT DATE(created_at) AS day, COUNT(*) AS messages FROM messages GROUP BY DATE(created_at) ORDER BY messages DESC LIMIT 1 """ ).fetchone() connection.close() return { "total_messages": total_messages, "total_users": total_users, "active_users": active_users, "top_users": top_users, "best_month": best_month, "best_day": best_day, } def get_user_statistics(user_id: int): connection = get_connection() user = connection.execute( """ SELECT full_name, username, messages, first_seen, last_seen FROM users WHERE user_id = ? """, (user_id,), ).fetchone() connection.close() return user # ========================================================= # KEYBOARDS # ========================================================= def main_keyboard(): return InlineKeyboardMarkup( inline_keyboard=[ [ InlineKeyboardButton( text="📊 Итоги 2026", callback_data="results", ) ], [ InlineKeyboardButton( text="👤 Мои итоги", callback_data="my_results", ) ], [ InlineKeyboardButton( text="🏆 Топ комьюнити", callback_data="top", ) ], ] ) def results_keyboard(): return InlineKeyboardMarkup( inline_keyboard=[ [ InlineKeyboardButton( text="🏆 Топ участников", callback_data="top", ) ], [ InlineKeyboardButton( text="👤 Мои итоги", callback_data="my_results", ) ], [ InlineKeyboardButton( text="🔙 Главное меню", callback_data="home", ) ], ] ) def back_keyboard(): return InlineKeyboardMarkup( inline_keyboard=[ [ InlineKeyboardButton( text="🔙 Главное меню", callback_data="home", ) ] ] ) # ========================================================= # START # ========================================================= @dp.message(CommandStart()) async def start_handler(message: Message): text = ( "🟣 VIPERR\n" "ИТОГИ 2026\n\n" "━━━━━━━━━━━━━━━━━━\n\n" "Один год.\n" "Одно комьюнити.\n" "Тысячи историй.\n\n" "Здесь собраны итоги нашего года:\n" "активность, участники и главные цифры.\n\n" "Готов посмотреть? 👀" ) await message.answer( text, reply_markup=main_keyboard(), ) # ========================================================= # СБОР СООБЩЕНИЙ ИЗ ГРУППЫ # ========================================================= @dp.message(F.chat.id == GROUP_ID) async def group_message_handler(message: Message): if not message.from_user: return # Игнорируем сообщения от ботов if message.from_user.is_bot: return full_name = ( message.from_user.full_name or message.from_user.username or "Unknown" ) save_message( user_id=message.from_user.id, username=message.from_user.username, full_name=full_name, ) # ========================================================= # ГЛАВНОЕ МЕНЮ # ========================================================= @dp.callback_query(F.data == "home") async def home_handler(callback: CallbackQuery): text = ( "🟣 VIPERR — ИТОГИ 2026\n\n" "━━━━━━━━━━━━━━━━━━\n\n" "Комьюнити — это не цифры.\n" "Но цифры помогают вспомнить,\n" "каким был этот год.\n\n" "Выбирай раздел 👇" ) await callback.message.edit_text( text, reply_markup=main_keyboard(), ) await callback.answer() # ========================================================= # ИТОГИ ГОДА # ========================================================= @dp.callback_query(F.data == "results") async def results_handler(callback: CallbackQuery): stats = get_statistics() total_messages = stats["total_messages"] total_users = stats["total_users"] active_users = stats["active_users"] # ----------------------------------------------------- # ЛУЧШИЙ МЕСЯЦ # ----------------------------------------------------- best_month = stats["best_month"] months = { 1: "Январь", 2: "Февраль", 3: "Март", 4: "Апрель", 5: "Май", 6: "Июнь", 7: "Июль", 8: "Август", 9: "Сентябрь", 10: "Октябрь", 11: "Ноябрь", 12: "Декабрь", } if best_month: month_number = int(best_month["month"]) best_month_text = ( f"{months.get(month_number, 'Неизвестно')} " f"— {best_month['messages']:,} сообщений" ) else: best_month_text = "Недостаточно данных" # ----------------------------------------------------- # ЛУЧШИЙ ДЕНЬ # ----------------------------------------------------- best_day = stats["best_day"] if best_day: best_day_text = ( f"{best_day['day']} " f"— {best_day['messages']:,} сообщений" ) else: best_day_text = "Недостаточно данных" # ----------------------------------------------------- # ЭКРАН # ----------------------------------------------------- text = ( "🟣 VIPERR — ИТОГИ 2026\n\n" "━━━━━━━━━━━━━━━━━━\n" f"💬 Сообщений\n" f"{total_messages:,}\n\n" f"👥 Участников\n" f"{total_users:,}\n\n" f"🔥 Активных участников\n" f"{active_users:,}\n\n" f"📅 Самый активный месяц\n" f"{best_month_text}\n\n" f"⚡ Самый активный день\n" f"{best_day_text}\n" "━━━━━━━━━━━━━━━━━━\n\n" "Это история, которую мы создали вместе 🟣" ) await callback.message.edit_text( text, reply_markup=results_keyboard(), ) await callback.answer() # ========================================================= # ТОП УЧАСТНИКОВ # ========================================================= @dp.callback_query(F.data == "top") async def top_handler(callback: CallbackQuery): stats = get_statistics() top_users = stats["top_users"] if not top_users: text = ( "🏆 ТОП VIPERR\n\n" "Пока недостаточно статистики.\n\n" "Как только появится активность,\n" "здесь появится рейтинг." ) else: medals = [ "🥇", "🥈", "🥉", "4️⃣", "5️⃣", "6️⃣", "7️⃣", "8️⃣", "9️⃣", "🔟", ] lines = [ "🏆 ТОП VIPERR", "", "Самые активные участники:", "", ] for index, user in enumerate(top_users): name = user["full_name"] if user["username"]: name = f"@{user['username']}" medal = medals[index] lines.append( f"{medal} {name}\n" f" 💬 {user['messages']:,} сообщений" ) lines.extend( [ "", "━━━━━━━━━━━━━━━━━━", "", "Спасибо каждому, кто делает", "VIPERR живым 🟣", ] ) text = "\n".join(lines) await callback.message.edit_text( text, reply_markup=back_keyboard(), ) await callback.answer() # ========================================================= # ЛИЧНЫЕ ИТОГИ # ========================================================= @dp.callback_query(F.data == "my_results") async def my_results_handler(callback: CallbackQuery): user = get_user_statistics( callback.from_user.id ) if not user: text = ( "👤 ТВОИ ИТОГИ\n\n" "Я пока не нашёл тебя в статистике.\n\n" "Напиши сообщение в VIPERR,\n" "и бот начнёт считать твою активность 🟣" ) else: first_seen = user["first_seen"][:10] text = ( "👤 ТВОИ ИТОГИ VIPERR\n\n" "━━━━━━━━━━━━━━━━━━\n\n" f"💬 Твоих сообщений\n" f"{user['messages']:,}\n\n" f"📅 В статистике с\n" f"{first_seen}\n\n" "━━━━━━━━━━━━━━━━━━\n\n" "Ты — часть истории VIPERR 🟣" ) await callback.message.edit_text( text, reply_markup=back_keyboard(), ) await callback.answer() # ========================================================= # ОБРАБОТКА НЕОЖИДАННЫХ ОШИБОК # ========================================================= @dp.errors() async def error_handler(event): logger.error( "Ошибка при обработке события: %s", event.exception, ) # ========================================================= # ЗАПУСК # ========================================================= async def main(): init_database() logger.info( "🟣 VIPERR Results Bot запущен" ) logger.info( "GROUP_ID: %s", GROUP_ID, ) await bot.delete_webhook( drop_pending_updates=True ) await dp.start_polling( bot, allowed_updates=dp.resolve_used_update_types(), ) # ========================================================= # MAIN # ========================================================= if __name__ == "__main__": try: asyncio.run(main()) except KeyboardInterrupt: logger.info( "Бот остановлен" )
